import json,urllib.request
B="https://firestore.googleapis.com/v1/projects/homestead-electric/databases/(default)/documents/jobs"
def un(v):
    for k in('stringValue','integerValue','doubleValue','booleanValue'):
        if k in v: return v[k]
    if 'arrayValue' in v: return [un(x) for x in v['arrayValue'].get('values',[])]
    if 'mapValue' in v: return {k:un(x) for k,x in v['mapValue'].get('fields',{}).items()}
out=[];tok=None
while True:
    d=json.load(urllib.request.urlopen(B+"?pageSize=300"+(f"&pageToken={tok}" if tok else ""),timeout=60))
    for doc in d.get('documents',[]):
        f=doc.get('fields',{}).get('data')
        if f: out.append(un(f))
    tok=d.get('nextPageToken')
    if not tok: break
json.dump(out,open('/tmp/jobs_cc.json','w'))
EOF
